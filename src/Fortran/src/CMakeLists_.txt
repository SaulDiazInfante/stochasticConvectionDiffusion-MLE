cmake_minimum_required(VERSION 3.15)
project(FortranModules LANGUAGES Fortran)

# Define Fortran Compiler
set(CMAKE_Fortran_COMPILER ifx)

# Set the MKL root directory
set(MKLROOT /opt/intel/mkl)

# Set MKL_DIR to the directory containing MKL configuration files
set(MKL_DIR ${MKLROOT}/lib/cmake/mkl)

# Add MKL to CMAKE_PREFIX_PATH
set(CMAKE_PREFIX_PATH ${MKL_DIR} ${CMAKE_PREFIX_PATH})

# Find MKL
find_package(MKL REQUIRED)
if(MKL_FOUND)
    message(STATUS "MKL found: ${MKL_INCLUDE_DIRS}")
else()
    message(FATAL_ERROR "MKL not found")
endif()

# Define source files (list modules first)
set(SRC_FILES
    src/mod_par_generators.f90
    src/mod_sde_coefficients.f90
    # src/mod_random_number_generator.f90
    # src/mkl_vsl.f90
    src/main.f90
)

# Set module output directory
set(CMAKE_Fortran_MODULE_DIRECTORY "${CMAKE_BINARY_DIR}/modules")

# Add include directories
include_directories(${MKL_INCLUDE_DIRS})
include_directories(${MKLROOT}/include)

# Add executable
add_executable(my_fortran_program ${SRC_FILES})

# Link MKL libraries
target_link_libraries(my_fortran_program ${MKL_LIBRARIES})
target_link_libraries(my_fortran_program ${MKLROOT}/lib/intel64/libmkl_intel_lp64.a ${MKLROOT}/lib/intel64/libmkl_sequential.a ${MKLROOT}/lib/intel64/libmkl_core.a -lpthread -lm -ldl)

# Add dependencies
add_dependencies(my_fortran_program 
    mod_par_generators
    mod_sde_coefficients
    # mod_random_number_generator
)
add_library(mod_par_generators OBJECT src/mod_par_generators.f90)
add_library(mod_sde_coefficients OBJECT src/mod_sde_coefficients.f90)
#add_library(mod_random_number_generator OBJECT src/mod_random_number_generator.f90)

# Include the module directory for compilation
target_include_directories(my_fortran_program PRIVATE ${CMAKE_Fortran_MODULE_DIRECTORY})

# Set compiler options to specify module directory
target_compile_options(my_fortran_program PRIVATE -J${CMAKE_Fortran_MODULE_DIRECTORY})
